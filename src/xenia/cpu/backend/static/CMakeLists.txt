add_library(xenia-cpu-backend-static STATIC static_backend.cc runtime.cc)
target_link_libraries(xenia-cpu-backend-static PUBLIC xenia-base xenia-cpu)
xe_target_defaults(xenia-cpu-backend-static)
if(MSVC)
  target_compile_options(xenia-cpu-backend-static PRIVATE /fp:strict)
else()
  target_compile_options(xenia-cpu-backend-static PRIVATE -fno-fast-math -ffp-contract=off -frounding-math)
endif()
if(XENIA_STATIC_MODULES_DIR)
  if(NOT EXISTS "${XENIA_STATIC_MODULES_DIR}/catalog.cmake")
    message(FATAL_ERROR "Generate a private AOT catalog with tools/static_recomp/link.py")
  endif()
  include("${XENIA_STATIC_MODULES_DIR}/catalog.cmake")
  target_sources(xenia-cpu-backend-static PRIVATE ${XENIA_AOT_SOURCES})
else()
  target_sources(xenia-cpu-backend-static PRIVATE catalog_empty.cc)
  if(XENIA_STATIC_ONLY)
    message(STATUS "Strict static build has no linked title modules; capture/negative tests only")
  endif()
endif()
