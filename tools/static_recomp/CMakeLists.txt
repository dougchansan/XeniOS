cmake_minimum_required(VERSION 3.20)
project(xenios_static_recomp_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(XENIOS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
option(AOT_SANITIZE "Enable undefined-behavior and address sanitizers" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(aot-runtime STATIC "${XENIOS_ROOT}/src/xenia/cpu/backend/static/runtime.cc")
target_include_directories(aot-runtime PUBLIC "${XENIOS_ROOT}/src" "${CMAKE_CURRENT_LIST_DIR}/tests")
if(MSVC)
  target_compile_options(aot-runtime PUBLIC /fp:strict /W4)
else()
  target_compile_options(aot-runtime PUBLIC -Wall -Wextra -fno-fast-math -ffp-contract=off -frounding-math)
  if(AOT_SANITIZE)
    target_compile_options(aot-runtime PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(aot-runtime PUBLIC -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_executable(aot-runtime-tests tests/runtime_tests.cc)
target_link_libraries(aot-runtime-tests PRIVATE aot-runtime)
add_test(NAME runtime COMMAND aot-runtime-tests)
add_test(NAME exporter COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/tests/test_export.py")
add_test(NAME ios-build-driver COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/tests/test_build_ios.py")
if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/fixtures/sources.cmake")
 include("${CMAKE_CURRENT_BINARY_DIR}/fixtures/sources.cmake")
 add_executable(aot-fixture-tests ${AOT_FIXTURE_SOURCES})
 target_link_libraries(aot-fixture-tests PRIVATE aot-runtime)
 if(NOT MSVC)
  target_compile_options(aot-fixture-tests PRIVATE -O0)
 endif()
 add_test(NAME ppc-fixtures COMMAND aot-fixture-tests)
endif()
